## Configuration Example

```yaml
strategy_engine_v2:
  enabled: true
  processing:
    concurrent_workers: 10
    processing_timeout: 30s
    max_historical_candles: 100
    batch_size: 50
  dataframe:
    library: "gota"
    memory_limit_mb: 512
    enable_parallel_processing: true
  strategies:
    registry:
      auto_discovery: true
      discovery_path: "./internal/strategy/v2/strategies"
    execution:
      parallel_strategies: true
      strategy_timeout: 10s
      error_handling: "continue_on_error"
  persistence:
    batch_insert: true
    batch_size: 100
    enable_audit_log: true
  monitoring:
    enable_metrics: true
    enable_tracing: true
    log_level: "info"
```

## Sample Strategy Usage

```go
// Create and configure strategy
strategy := strategies.NewBBWidthStrategyV2()
strategy.Configure(map[string]interface{}{
    "enabled": true,
    "parameters": map[string]interface{}{
        "period":   20,
        "std_dev":  2.0,
        "lookback": 5,
    },
})

// Register with registry
registry := v2.NewStrategyRegistryV2(config)
registry.Register(strategy)

// Process data
result, err := strategy.Process(dataFrame)
```
