## **Phase 1: Database Schema Creation for Parameter Persistence**

### **Phase 1.1: Schema Design & Migration** (Day 1)
**Objective**: Create database schema for strategy parameter persistence

#### **Tasks**:
1. **Design Parameter Storage Schema**
   ```sql
   -- Primary table for strategy parameters
   CREATE TABLE strategy_parameters_v2 (
       id VARCHAR(36) PRIMARY KEY,
       stock_id VARCHAR(36) NOT NULL,
       strategy_name VARCHAR(100) NOT NULL,
       timestamp TIMESTAMP NOT NULL,
       parameters JSONB NOT NULL,
       state_data JSONB,
       metadata JSONB,
       version INT DEFAULT 1,
       is_valid BOOLEAN DEFAULT true,
       is_synchronized BOOLEAN DEFAULT false,
       created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
       updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
       
       INDEX idx_stock_strategy (stock_id, strategy_name),
       INDEX idx_timestamp (timestamp),
       INDEX idx_strategy_timestamp (strategy_name, timestamp),
       INDEX idx_valid_sync (is_valid, is_synchronized)
   );
   
   -- State transitions audit table
   CREATE TABLE strategy_state_transitions (
       id VARCHAR(36) PRIMARY KEY,
       stock_id VARCHAR(36) NOT NULL,
       strategy_name VARCHAR(100) NOT NULL,
       from_state JSONB,
       to_state JSONB,
       transition_type VARCHAR(50) NOT NULL,
       reason VARCHAR(255),
       metadata JSONB,
       timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
       
       INDEX idx_stock_strategy_time (stock_id, strategy_name, timestamp)
   );
   ```

   ```go
   // internal/service/group_execution_scheduler.go
   type GroupExecutionSchedulerV2 struct {
       // ... existing fields
       v2Engine *v2.ProductionReadyEngineV2
   }
   
   func (s *GroupExecutionSchedulerV2) SetV2Engine(engine *v2.ProductionReadyEngineV2) {
       s.v2Engine = engine
   }
   
   func (s *GroupExecutionSchedulerV2) OnFiveMinClose(candle domain.Candle) {
       // Existing V1 logic
       s.processV1Strategies(candle)
       
       // New V2 engine processing
       if s.v2Engine != nil {
           s.processV2Strategies(candle)
       }
   }
   
   func (s *GroupExecutionSchedulerV2) processV2Strategies(candle domain.Candle) {
       ctx := context.Background()
       
       // Get active stock groups
       stockGroups, err := s.getActiveStockGroups()
       if err != nil {
           log.Error("Failed to get active stock groups: %v", err)
           return
       }
       
       if len(stockGroups) == 0 {
           log.Debug("No active stock groups for V2 processing")
           return
       }
       
       // Process with V2 engine
       startTime := time.Now()
       results, err := s.v2Engine.ProcessStockGroups(ctx, stockGroups, candle.Timestamp)
       processingTime := time.Since(startTime)
       
       if err != nil {
           log.Error("V2 engine processing failed: %v", err)
           return
       }
       
       log.Info("V2 engine processed %d stock groups in %v", 
           len(stockGroups), processingTime)
       
       // Log metrics
       metrics := s.v2Engine.GetMetrics()
       log.Info("V2 engine metrics: processed=%d, strategies=%d, avg_time=%v",
           metrics.TotalStocksProcessed, metrics.TotalStrategiesExecuted,
           metrics.AverageProcessingTime)
   }
   ```

   ```go
   // internal/strategy/v2/production_ready_engine.go
   func (engine *ProductionReadyEngineV2) fetchHistoricalData(
       ctx context.Context,
       stockGroups []domain.StockGroup,
       currentTime time.Time,
   ) (map[string]*dataframe.DataFrame, error) {
       stockDataFrames := make(map[string]*dataframe.DataFrame)
       
       // Get maximum required history across all strategies
       maxHistory := engine.getMaxRequiredHistory()
       
       for _, group := range stockGroups {
           for _, stock := range group.Stocks {
               // Calculate start time
               startTime := currentTime.Add(-time.Duration(maxHistory) * 5 * time.Minute)
               
               // Fetch candles using existing repository
               candles, err := engine.candleRepository.GetCandlesByTimeRange(
                   stock.StockID, "5min", startTime, currentTime,
               )
               if err != nil {
                   log.Error("Failed to fetch candles for %s: %v", stock.StockID, err)
                   continue
               }
               
               // Convert to DataFrame
               df, err := engine.convertCandlesToDataFrame(candles)
               if err != nil {
                   log.Error("Failed to convert candles for %s: %v", stock.StockID, err)
                   continue
               }
               
               stockDataFrames[group.ID] = df
               break // Use first stock's data for the group
           }
       }
       
       return stockDataFrames, nil
   }
   ```

**Objective**: Create strategy factory for dynamic strategy registration

   ```go
   // internal/strategy/v2/strategy_factory.go
   type StrategyFactory struct {
       paramsManager *StrategyParametersManager
   }
   
   func NewStrategyFactory(paramsManager *StrategyParametersManager) *StrategyFactory {
       return &StrategyFactory{
           paramsManager: paramsManager,
       }
   }
   
   func (f *StrategyFactory) CreateStrategy(strategyType string) (StrategyV2, error) {
       switch strategyType {
       case "1ST_ENTRY":
           return v2.NewFirstEntryStrategyV2Simple(f.paramsManager), nil
       case "2_30_ENTRY":
           return v2.NewTwoThirtyEntryStrategyV2(f.paramsManager), nil
       case "BB_WIDTH_ENTRY":
           return v2.NewBBWidthEntryStrategyV2(f.paramsManager), nil
       default:
           return nil, fmt.Errorf("unknown strategy type: %s", strategyType)
       }
   }
   
   func (f *StrategyFactory) RegisterAllStrategies(engine *ProductionReadyEngineV2) error {
       strategies := []string{"1ST_ENTRY", "2_30_ENTRY", "BB_WIDTH_ENTRY"}
       
       for _, strategyType := range strategies {
           strategy, err := f.CreateStrategy(strategyType)
           if err != nil {
               return fmt.Errorf("failed to create strategy %s: %w", strategyType, err)
           }
           
           if err := engine.RegisterStrategy(strategy); err != nil {
               return fmt.Errorf("failed to register strategy %s: %w", strategyType, err)
           }
           
           log.Info("Registered V2 strategy: %s", strategyType)
       }
       
       return nil
   }
   ```
